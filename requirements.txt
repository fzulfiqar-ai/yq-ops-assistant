# Production API image ONLY. Keep this lean — the free-tier host caps the
# container at 512 MB RAM, and every unused wheel eats into that headroom.
#
# The local Streamlit dashboard (dashboard/ui.py) and the dev-only file watcher
# live in requirements-dev.txt. They are NOT copied into the Docker image, so
# shipping streamlit/plotly here cost ~250 MB of image and RAM for nothing.

# --- web framework ---
fastapi
uvicorn[standard]
slowapi
python-multipart  # REQUIRED: FastAPI raises at import time on any File()/UploadFile
                  # route ("Form data requires python-multipart"). Nothing imports it
                  # by name, so a static import scan will NOT flag it. It used to
                  # arrive transitively via streamlit; do not remove it again.

# --- auth / crypto ---
PyJWT
cryptography
sqlglot           # app/sql_validator.py: the /ask allowlist walks a real Postgres parse tree
                  # (subqueries, CTEs, joins) instead of a FROM/JOIN regex. Pure Python.

# --- data layer ---
supabase
psycopg[binary]   # app/ingest_batch.py: the batch importer's OWN session on DATABASE_URL. PostgREST
                  # caps every RPC at the authenticator role's 8 s statement_timeout; a batch commit
                  # needs more, so the preview diff, ingest_commit and ingest_undo never go through it.
python-dotenv
requests
tenacity

# --- LLM ---
openai

pandas
openpyxl          # .xlsx  (app/catalog.py, app/order_sheet.py)
xlrd              # .xls   (legacy Focus ERP daily reports, via pandas.read_excel)
pdfplumber        # PO / invoice parsing
fpdf2             # app/catalog.py PDF export
pyyaml

pillow            # ad cards
numpy             # app/video_gen.py alpha-matting (also a pandas dep, pinned here
